# Sensi — Python dependencies (web workflow)
# Install with: pip install -r requirements.txt
#
# Every LLM provider (OpenAI, Anthropic, Google, Cloudflare, local) is reached
# through a single OpenAI-compatible client (langchain-openai → ChatOpenAI with a
# swapped base_url), so no provider-specific SDKs are needed here.

# ── LangGraph agent stack ────────────────────────────────────────────────────
langgraph
langchain-openai      # pulls in langchain-core transitively
httpx                 # Tavily web search + outbound HTTP
python-dotenv         # loads the .env at the repo root

# ── Web backend (FastAPI; serves the built React SPA in production) ───────────
fastapi
uvicorn[standard]
pydantic
itsdangerous            # signs the session cookie (starlette SessionMiddleware)

# ── Google sign-in + per-user persona storage ─────────────────────────────────
google-auth             # verifies Google ID tokens (GIS button); no client secret
google-cloud-firestore  # per-user personas in production (PERSONA_STORE=firestore)

# ── Spatial soundness (edit-tool placement + geometry validation) ─────────────
# Shapely powers sound placement of new windows / furniture (clear-span, buffer,
# representative_point). Self-contained wheels — no system libgeos needed.
shapely
